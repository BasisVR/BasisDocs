---
title: Miglior tempo sul server
description: Salvare il miglior giro di ogni giocatore sul server con BasisServerStorageShim, così lo segue tra un dispositivo e l'altro.
---

import { Callout } from 'fumadocs-ui/components/callout';

`BasisSaveShim` salva sulla macchina del giocatore. `BasisServerStorageShim` salva sul **server**, per giocatore, così un miglior tempo ottenuto su PC è ancora lì quando visita il mondo da Quest.

```csharp title="ServerBestTime.cs (Cilbox)"
using Basis.Shims;
using Cilbox;
using TMPro;
using UnityEngine;

[Cilboxable]
public class ServerBestTime : MonoBehaviour
{
    const string Key = "best-lap";
    public TextMeshPro label;
    float best = float.MaxValue;

    void Start()
    {
        if (!BasisServerStorageShim.IsAvailable(this)) { label.text = "Times are not saved on this server"; return; }
        BasisServerStorageShim.LoadString(this, Key, (result, text) =>
        {
            if (result == BasisServerStorageResult.Ok && float.TryParse(text, out float saved)) best = saved;
            Show();
        });
    }

    public void LapFinished(float seconds)
    {
        if (seconds >= best) return;
        best = seconds;
        Show();
        BasisServerStorageShim.SaveString(this, Key, best.ToString("F3"), result =>
        {
            if (result != BasisServerStorageResult.Ok) Debug.LogWarning($"best lap not saved: {result}");
        });
    }

    void Show() => label.text = best < float.MaxValue ? $"Your best: {best:F2}s" : "No lap yet";
}
```

<Callout type="info">
L'archiviazione è per giocatore e per server, con un totale ridotto (16 KB per impostazione predefinita) condiviso tra tutti i contenuti. Salva valori compatti, non log.
</Callout>
