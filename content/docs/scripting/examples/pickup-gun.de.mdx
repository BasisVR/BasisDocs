import { Callout } from 'fumadocs-ui/components/callout';

## Einrichtung

1. Modellieren oder importieren Sie eine Waffe. Fügen Sie einen Collider und einen **Rigidbody** hinzu.
2. Fügen Sie **Basis Pickup Interactable** hinzu.
3. Fügen Sie **Basis Pickup Sync Networking** hinzu und weisen Sie das Pickup zu. Dadurch werden die Position der Waffe und der Halter synchronisiert.
4. Fügen Sie eine **AudioSource** und ein **ParticleSystem** für das Mündungsfeuer hinzu sowie ein untergeordnetes Transform an der Mündung.
5. Fügen Sie **PickupGun** hinzu und weisen Sie alles zu.

## Skript

```csharp title="PickupGun.cs (Cilbox)"
using Basis;
using Basis.Network.Core;
using Basis.Scripts.BasisSdk.Interactions;
using Basis.Scripts.Device_Management.Devices;
using Basis.Shims;
using Cilbox;
using UnityEngine;

[Cilboxable]
public class PickupGun : MonoBehaviour
{
    public BasisPickupInteractable pickup;
    public Transform muzzle;
    public AudioSource shotSound;
    public ParticleSystem flash;
    public float range = 50f;
    public float cooldown = 0.2f;

    BasisNetworkShim net;
    bool held;
    float nextShot;

    void Start()
    {
        net = GetComponent<BasisNetworkShim>();
        net.NetworkMessageReceived += OnMessage;
        pickup.OnInteractStartEvent.AddListener(OnGrab);
        pickup.OnInteractEndEvent.AddListener(OnDrop);
        pickup.OnPickupUse.AddListener(OnUse);
    }

    void OnDestroy()
    {
        if (net != null) net.NetworkMessageReceived -= OnMessage;
    }

    void OnGrab(BasisInput input) { held = true; }
    void OnDrop(BasisInput input) { held = false; }

    void OnUse(BasisPickUpUseMode mode)
    {
        if (!held || mode != BasisPickUpUseMode.OnPickUpUseDown || Time.time < nextShot) return;
        nextShot = Time.time + cooldown;

        PlayShot();
        BasisHapticsShim.Play(this, BasisHapticHand.Both, 0.6f, 0.08f);
        if (net.HasNetworkID) net.SendCustomNetworkEvent(new byte[] { 1 }, DeliveryMethod.ReliableUnordered);

        if (Physics.Raycast(muzzle.position, muzzle.forward, out RaycastHit hit, range))
        {
            Debug.Log($"hit {hit.collider.name}");
        }
    }

    void OnMessage(ushort sender, byte[] buffer, DeliveryMethod method)
    {
        if (buffer != null && buffer.Length == 1 && buffer[0] == 1) PlayShot();
    }

    void PlayShot()
    {
        shotSound.Play();
        flash.Play();
    }
}
```

## Funktionsweise

- **Basis Pickup Sync Networking** übernimmt beim Greifen den Besitz und streamt die Pose der Waffe, sodass alle sie in der Hand des Halters sehen. Für diesen Teil ist kein Code nötig.
- **Der Schuss ist ein Ereignis**, kein Zustand. Niemand, der später beitritt, muss wissen, dass ein Schuss gefallen ist, daher gibt es keine Behandlung für später beitretende Spieler.
- **Nur der Schütze führt den Raycast aus.** Trefferergebnisse, die für das Gameplay relevant sind (Schaden, Punkte), sollten vom Schützen entschieden und als eigenes Ereignis gesendet oder wie bei der [Anzeigetafel](/de/docs/scripting/examples/scoreboard) an einen Besitzer weitergeleitet werden.

<Callout type="info">
Deaktivieren Sie **CanNetworkSteal** an der Sync-Komponente, wenn andere Spieler die Waffe niemandem aus der Hand ziehen können sollen.
</Callout>
