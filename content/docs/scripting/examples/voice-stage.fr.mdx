---
title: Scène vocale
description: Une scène qui diffuse la voix de l'orateur depuis de grands haut-parleurs, avec BasisVoiceRoutingShim.
---

```csharp title="VoiceStage.cs (Cilbox, world)"
using Basis.Scripts.Networking.NetworkedAvatar;
using Basis.Shims;
using Cilbox;
using UnityEngine;

[Cilboxable]
public class VoiceStage : MonoBehaviour
{
    public Collider stageArea;
    public Transform speaker;
    public float range = 60f;

    int route;
    ushort routedPlayer;
    float nextCheck;

    void Update()
    {
        if (Time.time < nextCheck) return;
        nextCheck = Time.time + 0.5f;

        BasisNetworkPlayer onStage = null;
        foreach (BasisNetworkPlayer p in BasisNetworkPlayer.GetAllPlayers())
        {
            if (p.Player != null && stageArea.bounds.Contains(p.Player.GetPosition())) { onStage = p; break; }
        }

        if (onStage != null && route != 0 && onStage.playerId == routedPlayer) return;
        Stop();
        if (onStage == null) return;
        route = BasisVoiceRoutingShim.RouteVoiceToObject(onStage, speaker, 2f, range);
        routedPlayer = onStage.playerId;
    }

    void Stop()
    {
        if (route != 0) BasisVoiceRoutingShim.StopVoiceRoute(route);
        route = 0;
    }

    void OnDestroy() => Stop();
}
```

- Chaque client route la voix localement à partir des positions qu'il connaît déjà. Il n'y a aucun trafic réseau supplémentaire.
- `RouteVoiceToObject` demande l'autorisation à l'orateur. Tant qu'il ne l'a pas accordée, ou s'il la refuse, il est entendu normalement.
- Vérifiez dans **Basis > Tools > Cilbox Permissions** que `BasisNetworkPlayer.GetAllPlayers` est autorisé dans votre sandbox.
