description: Стая из 200 птиц, дёшево анимируемая в Cilbox за счёт пакетной записи трансформов с помощью BasisTransformBatchShim.
---

Перемещение 200 трансформов по одному из скрипта Cilbox стоит сотни интерпретируемых вызовов за кадр. Пакетная обработка сводит их к одному.

```csharp title="Flock.cs (Cilbox)"
using Basis.Shims;
using Cilbox;
using UnityEngine;

[Cilboxable]
public class Flock : MonoBehaviour
{
    public Transform[] birds;
    public float radius = 10f;
    public float speed = 0.4f;

    float[] xyz;
    float[] phase;

    void Start()
    {
        xyz = new float[birds.Length * 3];
        phase = new float[birds.Length];
        for (int i = 0; i < phase.Length; i++) phase[i] = i * 0.37f;
    }

    void Update()
    {
        float t = Time.time * speed;
        Vector3 c = transform.position;
        float cx = c.x, cy = c.y, cz = c.z;
        for (int i = 0; i < birds.Length; i++)
        {
            float a = t + phase[i];
            xyz[i * 3] = cx + Mathf.Cos(a) * radius;
            xyz[i * 3 + 1] = cy + Mathf.Sin(a * 2f) * 2f;
            xyz[i * 3 + 2] = cz + Mathf.Sin(a) * radius;
        }
        BasisTransformBatchShim.SetPositions(birds, xyz);
    }
}
```

- Математика выполняется на простых float, что в интерпретаторе намного дешевле операторов `Vector3`.
- `SetPositions(Transform[], float[])` записывает всех птиц одним нативным вызовом.
- Используйте `BasisNetworkTimeShim.ServerTimeSeconds` вместо `Time.time`, если стая должна точно совпадать у всех игроков.
