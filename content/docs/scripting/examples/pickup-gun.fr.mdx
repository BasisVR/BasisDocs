import { Callout } from 'fumadocs-ui/components/callout';

## Configuration

1. Modélisez ou importez un pistolet. Ajoutez un collider et un **Rigidbody**.
2. Ajoutez **Basis Pickup Interactable**.
3. Ajoutez **Basis Pickup Sync Networking** et assignez le pickup. Cela synchronise la position du pistolet et la personne qui le tient.
4. Ajoutez une **AudioSource** et un **ParticleSystem** pour l'éclair de bouche, ainsi qu'un transform enfant au niveau de la bouche du canon.
5. Ajoutez **PickupGun** et assignez le tout.

## Script

```csharp title="PickupGun.cs (Cilbox)"
using Basis;
using Basis.Network.Core;
using Basis.Scripts.BasisSdk.Interactions;
using Basis.Scripts.Device_Management.Devices;
using Basis.Shims;
using Cilbox;
using UnityEngine;

[Cilboxable]
public class PickupGun : MonoBehaviour
{
    public BasisPickupInteractable pickup;
    public Transform muzzle;
    public AudioSource shotSound;
    public ParticleSystem flash;
    public float range = 50f;
    public float cooldown = 0.2f;

    BasisNetworkShim net;
    bool held;
    float nextShot;

    void Start()
    {
        net = GetComponent<BasisNetworkShim>();
        net.NetworkMessageReceived += OnMessage;
        pickup.OnInteractStartEvent.AddListener(OnGrab);
        pickup.OnInteractEndEvent.AddListener(OnDrop);
        pickup.OnPickupUse.AddListener(OnUse);
    }

    void OnDestroy()
    {
        if (net != null) net.NetworkMessageReceived -= OnMessage;
    }

    void OnGrab(BasisInput input) { held = true; }
    void OnDrop(BasisInput input) { held = false; }

    void OnUse(BasisPickUpUseMode mode)
    {
        if (!held || mode != BasisPickUpUseMode.OnPickUpUseDown || Time.time < nextShot) return;
        nextShot = Time.time + cooldown;

        PlayShot();
        BasisHapticsShim.Play(this, BasisHapticHand.Both, 0.6f, 0.08f);
        if (net.HasNetworkID) net.SendCustomNetworkEvent(new byte[] { 1 }, DeliveryMethod.ReliableUnordered);

        if (Physics.Raycast(muzzle.position, muzzle.forward, out RaycastHit hit, range))
        {
            Debug.Log($"hit {hit.collider.name}");
        }
    }

    void OnMessage(ushort sender, byte[] buffer, DeliveryMethod method)
    {
        if (buffer != null && buffer.Length == 1 && buffer[0] == 1) PlayShot();
    }

    void PlayShot()
    {
        shotSound.Play();
        flash.Play();
    }
}
```

## Fonctionnement

- **Basis Pickup Sync Networking** prend la propriété quand le pistolet est saisi et diffuse sa pose. Tout le monde le voit dans la main de celui qui le tient, sans aucun code.
- Le tir est un événement, pas un état. Personne qui rejoint plus tard n'a besoin de savoir qu'un tir a eu lieu, et il n'y a aucun code pour les joueurs arrivés en retard.
- Seul le tireur lance le raycast. Les résultats d'impact qui comptent pour le gameplay (dégâts, score) doivent être décidés par le tireur et envoyés sous forme d'événement propre, ou transmis à un propriétaire comme dans le [Tableau des scores](/fr/docs/scripting/examples/scoreboard).

<Callout type="info">
Désactivez **CanNetworkSteal** sur le composant de synchronisation si les autres joueurs ne doivent pas pouvoir arracher le pistolet de la main de quelqu'un.
</Callout>
