---
title: Lokalisiertes Schild
description: Ein Schild, das Text in der Sprache des Spielers anzeigt und mit BasisLanguageEventShim live umschaltet, wenn er sie ändert.
---

```csharp title="LocalizedSign.cs (Cilbox)"
using Basis.Shims;
using Cilbox;
using TMPro;
using UnityEngine;

[Cilboxable]
public class LocalizedSign : MonoBehaviour
{
    public TextMeshPro label;
    public string[] languageCodes = { "en", "de", "fr", "ja" };
    public string[] texts = { "Welcome", "Willkommen", "Bienvenue", "ようこそ" };

    void Start()
    {
        GetComponent<BasisLanguageEventShim>();
    }

    void OnLanguageChanged(string code)
    {
        label.text = texts[0];
        for (int i = 0; i < languageCodes.Length; i++)
        {
            if (code.StartsWith(languageCodes[i])) { label.text = texts[i]; return; }
        }
    }
}
```

- `OnLanguageChanged` wird sofort einmal mit der aktuellen Sprache aufgerufen.
- Durch den Vergleich mit `StartsWith` deckt `es` auch `es-MX` ab.
- Legen Sie die Arrays im Inspector statt mit einem Initialisierer fest, falls Ihre Sandbox Array-Initialisierer ablehnt, und stellen Sie sicher, dass die Schriftart die benötigten Glyphen enthält.
