---
title: パフォーマンススケーラー
description: BasisStatsShimとBasisGraphicsSettingsEventShimを使い、プレイヤーのフレームレートが下がったときに重いエフェクトを抑えます。
---

```csharp title="PerformanceScaler.cs (Cilbox)"
using Basis.Shims;
using Cilbox;
using UnityEngine;

[Cilboxable]
public class PerformanceScaler : MonoBehaviour
{
    public GameObject[] gpuHeavy;
    public GameObject[] cpuHeavy;
    float nextCheck;
    bool cutCpu, cutGpu, lowQuality;

    void Start()
    {
        GetComponent<BasisGraphicsSettingsEventShim>();
    }

    void OnGraphicsSettingsChanged(string qualityLevel, int qualityTier)
    {
        lowQuality = qualityTier <= BasisGraphicsSettingsShim.TierLow;
        Apply();
    }

    void Update()
    {
        if (Time.time < nextCheck) return;
        nextCheck = Time.time + 2f;
        float target = BasisStatsShim.TargetFramesPerSecond;
        if (target <= 0f) return;
        float fps = BasisStatsShim.FramesPerSecond;
        bool reduced = cutCpu || cutGpu;
        if (!reduced && fps < target * 0.8f)
        {
            int bottleneck = BasisStatsShim.Bottleneck;
            cutGpu = bottleneck == BasisStatsShim.BottleneckGpu;
            cutCpu = bottleneck == BasisStatsShim.BottleneckCpu;
            Apply();
        }
        else if (reduced && fps > target * 0.95f) { cutCpu = false; cutGpu = false; Apply(); }
    }

    void Apply()
    {
        foreach (GameObject g in gpuHeavy) g.SetActive(!(lowQuality || cutGpu));
        foreach (GameObject g in cpuHeavy) g.SetActive(!cutCpu);
    }
}
```

- 2つのしきい値（削減は80%、復元は95%）により、オンとオフが頻繁に切り替わるのを防ぎます。
- ボトルネックから、どちらを削るべきかが分かります。GPUならミラーやパーティクル、CPUなら群衆やスクリプトです。まだ計測中のときや、バランスが取れていると判定されたときは何も削らず、次のチェックで再試行します。
